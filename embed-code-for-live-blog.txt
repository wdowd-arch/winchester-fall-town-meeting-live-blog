<script>
  (function () {
    var origin='https://wdowd-arch.github.io';
    var wrap=document.getElementById('wtm-live-embed');
    var frame=document.getElementById('wtm-live-frame');
    if(!wrap||!frame)return;
    var connected=false;
    var timer=setTimeout(function(){if(!connected)wrap.classList.add('wtm-live-no-connect');},10000);
    function request(){try{frame.contentWindow.postMessage({type:'wtm-live-request-height'},origin);}catch(e){}}
    window.addEventListener('message',function(event){
      if(event.origin!==origin||event.source!==frame.contentWindow||!event.data||event.data.type!=='wtm-live-height')return;
      var height=Number(event.data.height);
      if(!isFinite(height)||height<=0)return;
      frame.style.height=Math.max(320,Math.ceil(height))+'px';
      connected=true;clearTimeout(timer);wrap.classList.remove('wtm-live-no-connect');
    });
    frame.addEventListener('load',request);
    request();
  })();
</script>
